## Exercise 4: Using @RequestBody

Question: Write a Spring Boot controller method that handles POST requests to "/echo". The method should accept a JSON object in the request body with a field "message", and return the same message. For example, if the request body is {"message": "Hello"}, it should return "Hello".

Answer:

```java
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class EchoController {
    
    @PostMapping("/echo")
    public String echo(@RequestBody Map<String, String> payload) {
        return payload.get("message");
    }
}
```

```go
package jsonparser

import (
	"encoding/json"
	"errors"
)

// ParseJSONToMap parses a JSON string into a map[string]interface{}.
// It returns an error if the JSON is invalid or not a map.
func ParseJSONToMap(jsonStr string) (map[string]interface{}, error) {
	if jsonStr == "" {
		// Return an empty map for empty JSON string
		return map[string]interface{}{}, nil
	}

	var data interface{}
	if err := json.Unmarshal([]byte(jsonStr), &data); err != nil {
		return nil, err
	}

	// Check if the data is a map[string]interface{}
	m, ok := data.(map[string]interface{})
	if !ok {
		return nil, errors.New("JSON is not a map")
	}

	return m, nil
}
```
