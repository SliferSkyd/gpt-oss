## Exercise 2: Using @RequestParam

Question: In a Spring Boot controller, write a method that handles GET requests to "/greet" and takes a request parameter "name". The method should return a greeting string that says "Hello, {name}!".

Answer:

```java
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class GreetController {
    
    @GetMapping("/greet")
    public String greet(@RequestParam String name) {
        return "Hello, " + name + "!";
    }
}
```

## Exercise 3: Using @PathVariable

Question: In a Spring Boot controller, write a method that handles GET requests to "/square/{number}" where {number} is a path variable. The method should return the square of the number.

Answer:

```java
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class SquareController {
    
    @GetMapping("/square/{number}")
    public int square(@PathVariable int number) {
        return number * number;
    }
}
```

```java
import org.springframework.web.bind.annotation.PostMapping;
```

## Exercise 7: Using @RequestPart

Question: In a Spring Boot controller, write a method that handles POST requests to "/upload". The method should accept a multipart request with a part named "file" which is a `MultipartFile`. The method should return the original filename of the uploaded file.

Answer:

```java
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestPart;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.multipart.MultipartFile;

@RestController
public class UploadController {
    
    @PostMapping("/upload")
    public String upload(@RequestPart("file") MultipartFile file) {
        return file.getOriginalFilename();
    }
}
```

```go
package jsonparser

import (
	"encoding/json"
	"errors"
)

// ParseJSONToMap parses a JSON string into a map[string]interface{}.
// It returns an error if the JSON is invalid or not a map.
func ParseJSONToMap(jsonStr string) (map[string]interface{}, error) {
	if jsonStr == "" {
		// Return an empty map for empty JSON string
		return map[string]interface{}{}, nil
	}

	var data interface{}
	if err := json.Unmarshal([]byte(jsonStr), &data); err != nil {
		return nil, err
	}

	// Check if the data is a map[string]interface{}
	m, ok := data.(map[string]interface{})
	if !ok {
		return nil, errors.New("JSON is not a map")
	}

	return m, nil
}
```

### 2. 使用 `iconv` 命令

如果你在 Linux 或 macOS 上，可以使用 `iconv` 命令来转换文件编码。下面是一个例子：

```bash
iconv -f utf-8 -t gbk input.txt > output.txt
```

这条命令将 `input.txt` 文件从 UTF-8 编码转换为 GBK 编码，并将结果输出到 `output.txt`。

```java
import java.io.*;

        // 写入新文件，使用 GBK 编码

            writer.newLine();
        }

        reader.close();
        writer.close();
    }
}
```
